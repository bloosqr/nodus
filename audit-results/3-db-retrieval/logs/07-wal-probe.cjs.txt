const Database = require('/home/user/nodus/node_modules/better-sqlite3');
const fs = require('fs');
for (const f of fs.readdirSync('.')) if (/\.sqlite/.test(f)) fs.unlinkSync(f);
const make = (file, label, n) => { const db = new Database(file); db.pragma('journal_mode = WAL'); db.exec('CREATE TABLE t(id INTEGER PRIMARY KEY, label TEXT)'); const ins = db.prepare('INSERT INTO t(label) VALUES (?)'); db.transaction(() => { for (let i = 0; i < n; i++) ins.run(label); })(); db.close(); };
make('imported.sqlite', 'imported', 50);
// The live vault: the main connection and, beside it, a long-lived read-only connection as the vector-scan worker keeps.
const live = 'vault.sqlite';
make(live, 'old', 10);
let main = new Database(live); main.pragma('journal_mode = WAL'); main.pragma('wal_autocheckpoint = 1000');
const reader = new Database(live, { readonly: true, fileMustExist: true });
console.log('reader sees', reader.prepare('SELECT COUNT(*) n FROM t').get().n);
main.transaction(() => { for (let i = 0; i < 200; i++) main.prepare('INSERT INTO t(label) VALUES (?)').run('old-after'); })();
// replaceDbFile: closeDb(); copyFileSync(source, target); openDatabase(target)
main.close();
console.log('wal left behind:', fs.existsSync(live + '-wal') ? fs.statSync(live + '-wal').size + ' bytes' : 'none');
fs.copyFileSync('imported.sqlite', live);
main = new Database(live);
try {
  console.log('after import, main sees', JSON.stringify(main.prepare('SELECT label, COUNT(*) n FROM t GROUP BY label').all()));
  console.log('integrity:', main.pragma('integrity_check', { simple: true }));
} catch (error) { console.log('after import, main fails:', error.message); }
try { console.log('worker connection sees', JSON.stringify(reader.prepare('SELECT label, COUNT(*) n FROM t GROUP BY label').all())); } catch (error) { console.log('worker connection fails:', error.message); }
